// Close to Close Volatility
f_coc(x, period, sqrtAnnual) =>
    mean = ta.sma(x, period)
    s = array.new_float(0)
    for i = 0 to period - 1 by 1
        array.push(s, math.pow(x[i] - mean, 2))
    sqrtAnnual * math.sqrt(array.sum(s) / (period - 1))

// Parkinson Volatility
f_park(period, sqrtAnnual) =>
    var LOG2 = math.log(2)
    powLogHighLow = math.pow(math.log(high / low), 2)
    sqrtAnnual * math.sqrt(1.0 / period * math.sum(1.0 / (4.0 * LOG2) * powLogHighLow, period))

// Garman Klass Volatility
f_gk(period, sqrtAnnual) =>
    var LOG2 = math.log(2)
    var SQRT_1_PERIOD = math.sqrt(1 / period)
    powLogHighLow = math.pow(math.log(high / low), 2)
    powLogCloseOpen = math.pow(math.log(close / open), 2)
    tmp = 0.5 * powLogHighLow - (2.0 * LOG2 - 1.0) * powLogCloseOpen
    sqrtAnnual * math.sqrt(math.sum(tmp, period)) * SQRT_1_PERIOD

// Rogers Satchell Volatility
f_rsv(period, sqrtAnnual) =>
    tmp = math.log(high / close) * math.log(high / open) + math.log(low / close) * math.log(low / open)
    sqrtAnnual * math.sqrt(math.sum(tmp, period) / period)

// Garman Klass Yang Zhang Extension Volatility
f_gkyz(period, sqrtAnnual) =>
    var LOG2 = math.log(2)
    var SQRT_1_PERIOD = math.sqrt(1 / period)
    powLogHighLow = math.pow(math.log(high / low), 2)
    powLogCloseOpen = math.pow(math.log(close / open), 2)
    lastClose = nz(close[1], close)
    powLogOpenClose1 = math.pow(math.log(open / lastClose), 2)
    tmp = powLogOpenClose1 + 0.5 * powLogHighLow - (2.0 * LOG2 - 1.0) * powLogCloseOpen
    sqrtAnnual * math.sqrt(math.sum(tmp, period)) * SQRT_1_PERIOD

// Yang Zhang Volatility
f_yz(a, period, sqrtAnnual) =>
    oaman = math.log(open) - math.log(nz(close[1], close))
    u = math.log(high) - math.log(open)
    d = math.log(low) - math.log(open)
    caman = math.log(close) - math.log(open)
    nMinusOne = period - 1
    avgo = ta.sma(oaman, period)
    avgc = ta.sma(caman, period)
    so = array.new_float(0)
    sc = array.new_float(0)
    for i = 0 to period - 1 by 1
        array.push(so, math.pow(oaman[i] - avgo, 2))
        array.push(sc, math.pow(caman[i] - avgc, 2))
    sumo = array.sum(so)
    sumc = array.sum(sc)
    Vo = sumo / nMinusOne
    Vc = sumc / nMinusOne
    Vrs = math.sum(u * (u - caman) + d * (d - caman), period) / period
    k = (a - 1.0) / (a + (period + 1.0) / nMinusOne)
    sqrtAnnual * math.sqrt(Vo + k * Vc + (1.0 - k) * Vrs)

// Exponentially Weighted Volatility
f_ewma(source, period, sqrtAnnual) =>
    var lambda = (period - 1) / (period + 1)
    squared = math.pow(source, 2)
    float v = na
    v := lambda * nz(v[1], squared) + (1.0 - lambda) * squared
    sqrtAnnual * math.sqrt(v)

// Mean Absolute Deviation (Adjusted)
f_mad(source, period, sqrtAnnual) =>
    var SQRT_HALF_PI = math.sqrt(math.asin(1))
    mean = ta.sma(source, period)
    S = array.new_float(0)
    for i = 0 to period - 1 by 1
        array.push(S, math.abs(source[i] - mean))
    sumS = array.sum(S)
    sqrtAnnual * (sumS / period) * SQRT_HALF_PI

// Median Absolute Deviation
f_mead(source, period, sqrtAnnual) =>
    median = ta.percentile_nearest_rank(source, period, 50)
    E = 0.0
    for i = 0 to period - 1 by 1
        E := E + math.abs(source[i] - median)
        E
    sqrtAnnual * math.sqrt(2) * (E / period)

            else if stuff.top < current.top and stuff.btm > current.btm

                del
            else if stuff.top > current.top and stuff.btm < current.btm
                del := true
                del
            else if stuff.top < current.top and stuff.top > current.btm
                del := true
                del
            if del

                if not na(stuff.bxActUp)

                if not na(stuff.bxActDn)

                    line.delete(stuff.midline)

    int(sizeScore + volScore)

// Score → grade label + color
getScoreGrade(score) =>
    grade = score >= 75 ? 'STRONG' : score >= 50 ? 'MODERATE' : 'WEAK'
    grade

getScoreColor(score) =>
    score >= 75 ? color.new(#00ffbb, 0) : score >= 50 ? color.new(#f5a623, 0) : color.new(#ff1100, 0)

// Gap fill percentage
getGapFill(top, bot, isBull) =>
    gapSize = top - bot
    result = 0
    if gapSize <= 0
        result := 100
        result
    else if isBull
        filled = math.max(0.0, top - math.max(low, bot))
        result := math.min(int(math.round(filled / gapSize * 100)), 100)
        result
    else
        filled = math.max(0.0, math.min(high, top) - bot)
        result := math.min(int(math.round(filled / gapSize * 100)), 100)
        result
    result

// Build label text
buildLabel(dir, score, age, fillPct) =>
    txt = dir
    if showScore
        txt := txt + '\n⚡' + str.tostring(score) + '%'
        txt
    if showAge
        txt := txt + '\n🕐' + str.tostring(age) + 'b'
        txt
    if showFill
        txt := txt + '\n📊' + str.tostring(fillPct) + '% fill'
        txt
    txt

method BullCleaner(FVGData d) =>
    if d.fvg.size() > 0
        for i = d.fvg.size() - 1 to 0 by 1
            // skip already mitigated
            if d.mitigated.get(i)
                continue

            top = d.gapTop.get(i)
            bot = d.gapBot.get(i)
            midP = math.avg(top, bot)

            // FIX-C: compute bool first, then use it — no if/else returning bool inside assignment
            condBodyClose = mit ? close < midP : close <= bot
            condWick = mit ? low < midP : low <= bot
            mitigateNow = bodyClose ? condBodyClose : condWick

            if mitigateNow
                d.mitigated.set(i, true)
                // FIX-A: increment via array
                array.set(_filledBull, 0, array.get(_filledBull, 0) + 1)

                // FIX: separate if blocks (not if/else) — avoids void vs bool mismatch
                if keepHistory
                    d.fvg.get(i).set_bgcolor(MitCol)
                    d.fvg.get(i).set_border_color(MitCol)
                    d.bvBox.get(i).set_bgcolor(color.new(color.gray, 95))
                    d.bvBox.get(i).set_border_color(color.new(color.gray, 95))
                    d.svBox.get(i).set_bgcolor(color.new(color.gray, 95))
                    d.svBox.get(i).set_border_color(color.new(color.gray, 95))
                    d.midLine.get(i).set_color(color.new(color.gray, 70))
                    // Score label: fade text color on mitigation
                    d.scoreLbl.get(i).set_textcolor(color.new(color.gray, 50))
                    d.scoreLbl.get(i).set_text('✓ MIT')
                if not keepHistory
                    d.fvg.get(i).delete()
                    d.midLine.get(i).delete()
                    d.bvBox.get(i).delete()
                    d.svBox.get(i).delete()
                    d.lbl.get(i).delete()
                    d.scoreLbl.get(i).delete()
                    d.fvg.remove(i)
                    d.midLine.remove(i)
                    d.bvBox.remove(i)
                    d.svBox.remove(i)
                    d.lbl.remove(i)
                    d.scoreLbl.remove(i)
                    d.birthBar.remove(i)
                    d.gapTop.remove(i)
                    d.gapBot.remove(i)
                    d.mitigated.remove(i)

method BearCleaner(FVGData d) =>

// ~~ LIVE LABEL UPDATER
// ════════════════════════════════════════════════════════════════════════

method UpdateLabels(FVGData d, isBull) =>
    if d.fvg.size() > 0
        for i = 0 to d.fvg.size() - 1 by 1
            if d.mitigated.get(i)
                continue
            top = d.gapTop.get(i)
            bot = d.gapBot.get(i)
            birth = d.birthBar.get(i)
            age = bar_index - birth
            fillPct = getGapFill(top, bot, isBull)
            dir = isBull ? '▲ FVG' : '▼ FVG'
            score = getFVGScore(top - bot, math.max(top - bot, 0.0001), 60, 40)
            txt = buildLabel(dir, score, age, fillPct)
            d.lbl.get(i).set_text(txt)

// ~~ MAIN FVG DETECTION
// ════════════════════════════════════════════════════════════════════════

FVG(l, l2, h, h2, c1, LBack, L2Back, HBack, H2Back, wick, avgFVG, buyVol, sellVol) =>

    bullGapSize = l - h2
    bearGapSize = l2 - h

    BullFVG = l > h2 and c1 > h2
    BearFVG = h < l2 and c1 < l2

    // ── Bull FVG ──────────────────────────────────────────────────────
    if BullFVG and not BullFVG[1] and wick >= avgFVG * BodySens
        array.set(_totalBull, 0, array.get(_totalBull, 0) + 1)

        future = getBarTime(HBack, extendLength)
        midVal = math.avg(l, h2)
        dist = future - H2Back
        score = getFVGScore(bullGapSize, avgFVG, buyVol, sellVol)
        txt = buildLabel('▲ FVG', score, 0, 0)

        b = box.new(H2Back, l, future, h2, bgcolor = Col, border_color = Col, extend = extend.none, xloc = xloc.bar_time)
        m = line.new(H2Back, midVal, future, midVal, xloc = xloc.bar_time, color = color.white, style = line.style_dashed, width = 1)
        bb = box.new(H2Back, midVal, int(H2Back + dist * (buyVol / 100)), l, bgcolor = UpCol, border_color = UpCol, extend = extend.none, xloc = xloc.bar_time)
        sb = box.new(H2Back, h2, int(H2Back + dist * (sellVol / 100)), midVal, bgcolor = DnCol, border_color = DnCol, extend = extend.none, xloc = xloc.bar_time)
        lb = label.new(future, midVal, txt, xloc = xloc.bar_time, style = label.style_label_left, color = color.new(color.white, 100), textcolor = color.new(#00ffbb, 0), size = size.normal)

        // ── Confluence Score Label (inside box, centered) ──
        scoreTxt = str.tostring(score) + '%\n' + getScoreGrade(score)
        scoreColor = getScoreColor(score)
        scoreX = int(H2Back + (future - H2Back) / 2)
        slb = label.new(scoreX, midVal, scoreTxt, xloc = xloc.bar_time, style = label.style_label_center, color = color.new(color.black, 75), textcolor = scoreColor, size = size.small)

        bullData.fvg.unshift(b)
        bullData.midLine.unshift(m)
        bullData.bvBox.unshift(bb)
        bullData.svBox.unshift(sb)
        bullData.lbl.unshift(lb)
        bullData.birthBar.unshift(bar_index)
        bullData.gapTop.unshift(l)
        bullData.gapBot.unshift(h2)
        bullData.mitigated.unshift(false)
        bullData.scoreLbl.unshift(slb)

        alert('🟢 Bull FVG Formed | ' + syminfo.ticker + ' | Score: ' + str.tostring(score) + '%', alert.freq_once_per_bar)

    // ── Bear FVG ──────────────────────────────────────────────────────
    if BearFVG and not BearFVG[1] and wick >= avgFVG * BodySens
        array.set(_totalBear, 0, array.get(_totalBear, 0) + 1)

        future = getBarTime(LBack, extendLength)
        midVal = math.avg(l2, h)
        dist = future - L2Back
        score = getFVGScore(bearGapSize, avgFVG, buyVol, sellVol)
        txt = buildLabel('▼ FVG', score, 0, 0)

        b = box.new(L2Back, l2, future, h, bgcolor = Col, border_color = Col, extend = extend.none, xloc = xloc.bar_time)
        m = line.new(L2Back, midVal, future, midVal, xloc = xloc.bar_time, color = color.white, style = line.style_dashed, width = 1)
        bb = box.new(L2Back, midVal, int(L2Back + dist * (buyVol / 100)), l2, bgcolor = UpCol, border_color = UpCol, extend = extend.none, xloc = xloc.bar_time)
        sb = box.new(L2Back, h, int(L2Back + dist * (sellVol / 100)), midVal, bgcolor = DnCol, border_color = DnCol, extend = extend.none, xloc = xloc.bar_time)
        lb = label.new(future, midVal, txt, xloc = xloc.bar_time, style = label.style_label_left, color = color.new(color.white, 100), textcolor = color.new(#ff1100, 0), size = size.normal)

        // ── Confluence Score Label (inside box, centered) ──
        scoreTxt = str.tostring(score) + '%\n' + getScoreGrade(score)
        scoreColor = getScoreColor(score)
        scoreX = int(L2Back + (future - L2Back) / 2)
        slb = label.new(scoreX, midVal, scoreTxt, xloc = xloc.bar_time, style = label.style_label_center, color = color.new(color.black, 75), textcolor = scoreColor, size = size.small)

        bearData.fvg.unshift(b)
        bearData.midLine.unshift(m)
        bearData.bvBox.unshift(bb)
        bearData.svBox.unshift(sb)
        bearData.lbl.unshift(lb)
        bearData.birthBar.unshift(bar_index)
        bearData.gapTop.unshift(l2)
        bearData.gapBot.unshift(h)
        bearData.mitigated.unshift(false)
        bearData.scoreLbl.unshift(slb)

        alert('🔴 Bear FVG Formed | ' + syminfo.ticker + ' | Score: ' + str.tostring(score) + '%', alert.freq_once_per_bar)
