// ─────────────────────────────────────────────────────────────────────────────
// RMSD Function (LyroRS)
// ─────────────────────────────────────────────────────────────────────────────
rmsd(src, benchmark, length) =>
    sum_sq = 0.0
    for i = 0 to length - 1
        d = src[i] - benchmark[i]
        sum_sq += d * d
    math.sqrt(sum_sq / length)

calculateStrength(float obRange, float breakoutDist, float relVol) =>
    float dispFactor = math.min(breakoutDist / (obRange * 5), 1.0)
    float volFactor = math.min(relVol / 2.0, 1.0)
    float score = (dispFactor * 0.6 + volFactor * 0.4) * 100
    math.round(score, 1)

getBarTime(t_, barsOffset_) =>
    t_ + barsOffset_ * timeframe.in_seconds(timeframe.period) * 1000

BarsBack(val_, hl_) =>
    back_ = time
    for i = 0 to 4999
        if val_ == hl_[i]
            back_ := time[i]
            break
    back_

VolumePower(v_, c_, l_, h_, ca_) =>
    r_ = h_ - l_
    r_ == 0 ? 50 : ca_ ?
     math.round(math.round(v_ * (c_ - l_) / r_) / v_ * 100) :
     math.round(math.round(v_ * (h_ - c_) / r_) / v_ * 100)

getScoreGrade(score_) =>
    score_ >= 75 ? 'STRONG' : score_ >= 50 ? 'MODERATE' : 'WEAK'

getScoreColor(score_) =>
    score_ >= 75 ? color.new(#00ffbb, 0) : score_ >= 50 ? color.new(#f5a623, 0) : color.new(#ff1100, 0)

getConfScore(gapSize_, avgWick_, buyVol_, sellVol_) =>
    sizeScore_ = math.min(gapSize_ / math.max(avgWick_, 0.0001) * 50, 50)
    volScore_  = math.min(math.abs(buyVol_ - sellVol_) / 2, 50)
    int(sizeScore_ + volScore_)

method OBMergeBull(OBData d, newTop_, newBot_, HBack_, buyVol_, sellVol_, avgWick_) =>
    merged_ = false
    if d.ob.size() > 0 and not d.mitigated.get(0)
        existTop_ = d.obTop.get(0)
        existBot_ = d.obBot.get(0)
        if isOBOverlap(newTop_, newBot_, existTop_, existBot_)
            mergedTop_ = math.max(existTop_, newTop_)
            mergedBot_ = math.min(existBot_, newBot_)
            midVal_    = math.avg(mergedTop_, mergedBot_)
            future_    = getBarTime(HBack_, ob_extLen)
            dist_      = future_ - HBack_
            score_     = getConfScore(mergedTop_ - mergedBot_, avgWick_, buyVol_, sellVol_)
            scoreTxt_  = str.tostring(score_) + '%\n' + getScoreGrade(score_)
            scoreX_    = int(HBack_ + dist_ / 2)

            d.ob.get(0).set_top(mergedTop_)
            d.ob.get(0).set_bottom(mergedBot_)
            d.ob.get(0).set_right(future_)

            d.midLine.get(0).set_y1(midVal_)
            d.midLine.get(0).set_y2(midVal_)
            d.midLine.get(0).set_x2(future_)

            d.bvBox.get(0).set_top(mergedTop_)
            d.bvBox.get(0).set_bottom(midVal_)
            d.bvBox.get(0).set_right(int(HBack_ + dist_ * (buyVol_ / 100)))

            d.svBox.get(0).set_top(midVal_)
            d.svBox.get(0).set_bottom(mergedBot_)
            d.svBox.get(0).set_right(int(HBack_ + dist_ * (sellVol_ / 100)))

            d.lbl.get(0).set_y(midVal_)

            d.scoreLbl.get(0).set_x(scoreX_)
            d.scoreLbl.get(0).set_y(midVal_)
            d.scoreLbl.get(0).set_text(scoreTxt_)
            d.scoreLbl.get(0).set_textcolor(getScoreColor(score_))

            d.obTop.set(0, mergedTop_)
            d.obBot.set(0, mergedBot_)
